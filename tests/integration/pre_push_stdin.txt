-- deletion-guard.yml --
output:
  - execution_out
pre-push:
  jobs:
    - run: echo PLAIN > plain.ran
    - run: echo {push_files} > files.ran
    - run: cat > filtered.stdin
      use_stdin: true
      glob: '*.md'
    - run: cat > custom.stdin
      use_stdin: true
      files: 'printf ""'
    - run: sh guard.sh
      use_stdin: true

-- guard.sh --
#!/bin/sh
cat > guard.stdin
while read -r local_ref local_oid remote_ref remote_oid; do
  if [ "$local_oid" = 0000000000000000000000000000000000000000 ]; then
    echo "PROTECTED $remote_ref"
    exit 1
  fi
done < guard.stdin
