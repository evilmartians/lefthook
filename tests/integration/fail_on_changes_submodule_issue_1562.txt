# A staged submodule and an untracked symlink to a directory must not break the changeset
lefthook run pre-commit
! stderr 'Unable to hash'

# Changes made by a hook are still detected next to a submodule
! lefthook run modify-file
stderr 'Error: files were modified by a hook, and fail_on_changes is enabled'

-- app/lefthook.yml --
pre-commit:
  fail_on_changes: always
  jobs:
    - run: "true"

modify-file:
  fail_on_changes: always
  jobs:
    - run: echo newline >> file.txt

-- app/file.txt --
